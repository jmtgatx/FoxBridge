SELECT ALLTRIM(a.sono) AS sono, ALLTRIM(a.custno) AS custno, ALLTRIM(a.company) AS company, ALLTRIM(a.address1) AS address1, ALLTRIM(a.address2) AS address2, ALLTRIM(a.city) AS city, ALLTRIM(a.state) AS state, ALLTRIM(a.zip) AS zip, ALLTRIM(a.country) AS country, ALLTRIM(a.phone) AS phone, ALLTRIM(a.faxno) AS faxno, ALLTRIM(b.email) AS email, ALLTRIM(a.ponum) AS ponum, ALLTRIM(a.upsaccount) AS upsaccount, IIF(a.billoption == "SHP", "Shipper", IIF(a.billoption == "REC", "Receiver", IIF(a.billoption == "TP", "Third Party", a.billoption))) AS billoption, ALLTRIM(a.companyb) AS companyb, ALLTRIM(a.address1b) AS address1b, ALLTRIM(a.address2b) AS address2b, ALLTRIM(a.cityb) AS cityb, ALLTRIM(a.stateb) AS stateb, ALLTRIM(a.zipb) AS zipb, ALLTRIM(a.sysco) AS sysco, ALLTRIM(a.ref1) AS ref1, ALLTRIM(a.ref2) AS ref2, ALLTRIM(a.qvnopt) AS qvnopt, ALLTRIM(a.service) AS service, ALLTRIM(a.codopt) AS codopt, ALLTRIM(a.saturday) AS saturday, ALLTRIM(a.cshipno) AS cshipno, ALLTRIM(a.dept) AS dept FROM soshad01 a INNER JOIN aradrs01 b ON (a.custno = b.custno AND a.company = b.company) ORDER BY a.sono DESC into table UPS_soshad01.dbf
